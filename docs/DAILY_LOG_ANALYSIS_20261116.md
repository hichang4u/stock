**이벤트 루프 정지 찾기(하트비트 간격, 틱이 밀려 들어온 묶음).**

```bash
cd /d/Private/stock && PYTHONIOENCODING=utf-8 ./.venv/Scripts/python.exe -c "
import glob, json
from datetime import datetime
from src.utils.jsonl_gz import read_jsonl_gz
D='20261116'; TK='010170'
hb=[datetime.fromisoformat(json.loads(l)['ts']) for l in open(f'D:/Private/stock-prod/data/logs/{D}.jsonl',encoding='utf-8') if '\"F4_HEARTBEAT\"' in l]
late=[(a,(b-a).total_seconds()) for a,b in zip(hb,hb[1:]) if (b-a).total_seconds()>35]
print('하트비트 35초 초과', len(late), '| 최대', round(max(s for _,s in late),1) if late else 0, '|', [a.strftime('%H:%M:%S') for a,_ in late])
rows=[]
for p in sorted(glob.glob(f'D:/Private/stock-prod/data/strategy_ticks/{D}/{TK}.*.jsonl.gz')): rows+=read_jsonl_gz(p).rows
ws=sorted((r for r in rows if r.get('source')=='ws' and r.get('source_ts')), key=lambda r:r['received_at'])
prev=None
for r in ws:
    t=datetime.fromisoformat(r['received_at']); lag=(t-datetime.fromisoformat(r['source_ts'])).total_seconds()
    if lag>10 and prev and (t-prev).total_seconds()>3:
        print('무수신', prev.time(), '->', t.time(), round((t-prev).total_seconds(),1), '초 | 수신-거래소', round(lag,1), '초')
    prev=t
"
```
